cmake_minimum_required(VERSION 3.16)
project(dvrmc_portable LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DVRMC_SANITIZE "Enable undefined-behaviour/address sanitizers" OFF)
if(DVRMC_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(dvrmc_core src/control.cpp src/device.cpp)
target_include_directories(dvrmc_core PUBLIC include)
add_library(dvrmc_host src/host.cpp)
target_link_libraries(dvrmc_host PUBLIC dvrmc_core)
foreach(target dvrmc_core dvrmc_host)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_executable(controller_tests tests/controller_tests.cpp)
target_link_libraries(controller_tests PRIVATE dvrmc_host)
add_test(NAME controller COMMAND controller_tests)
add_executable(simulate examples/simulate.cpp)
target_link_libraries(simulate PRIVATE dvrmc_host)
add_executable(top_tests tests/top_tb.cpp src/top.cpp)
target_link_libraries(top_tests PRIVATE dvrmc_core)
add_test(NAME hls_top_native COMMAND top_tests)

add_test(NAME two_motor_model COMMAND simulate)

add_executable(virtual_motor examples/virtual_motor.cpp)
target_link_libraries(virtual_motor PRIVATE dvrmc_host)
add_executable(plant_tests tests/plant_tests.cpp)
target_link_libraries(plant_tests PRIVATE dvrmc_host)
add_executable(plant_top_tests tests/plant_top_tb.cpp src/top.cpp)
target_link_libraries(plant_top_tests PRIVATE dvrmc_core)
foreach(target virtual_motor plant_tests plant_top_tests)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
add_test(NAME virtual_plant COMMAND plant_tests)
add_test(NAME virtual_plant_hls_top COMMAND plant_top_tests)
add_test(NAME virtual_scenarios COMMAND virtual_motor ${CMAKE_CURRENT_BINARY_DIR}/virtual_motor.csv)
